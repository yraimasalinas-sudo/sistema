// ==================== REPORTES Y CONFIGURACIÓN ==================== function generateReports() { // 1. Obtener rango de fechas (Desde / Hasta) const fromInput = document.getElementById('rep-from-period')?.value || ''; // Formato: YYYY-MM const toInput = document.getElementById('rep-to-period')?.value || document.getElementById('rep-period')?.value || new Date().toISOString().slice(0, 7); const periodLabel = document.getElementById('rep-ord-period-label'); if (periodLabel) { periodLabel.innerText = fromInput ? `Periodo: ${fromInput} hasta ${toInput}` : `Periodo: Hasta ${toInput}`; } // ========================================== // SECCIÓN 1: REPORTE DE CUOTAS MENSUALES (ORDINARIO) // ========================================== const ordBody = document.getElementById('report-ordinary-body'); if (ordBody) { ordBody.innerHTML = ''; // Filtrar gastos ordinarios en el rango seleccionado const ordinaryExpenses = state.expenses.filter(e => { const isOrd = e.type === 'ordinaria' || e.type === 'ordinary'; if (!isOrd) return false; if (!e.period) return true; if (fromInput && e.period < fromInput) return false; if (toInput && e.period > toInput) return false; return true; }); state.owners.forEach(o => { const factor = o.alicuota > 1 ? o.alicuota / 100 : Number(o.alicuota || 0); let totalAsignado = 0; let totalPagado = 0; let mesesAdeudadosCount = 0; let conceptosPendientes = []; ordinaryExpenses.forEach(exp => { const montoAsignadoItem = Number(exp.amount || 0) * factor; totalAsignado += montoAsignadoItem; const approvedPayments = state.payments.filter(p => p.apto === o.apto && p.status === 'Aprobado' && (p.type === 'ordinaria' || p.type === 'ordinary') && (!p.period || p.period === exp.period) ); const pagadoItem = approvedPayments.reduce((acc, p) => acc + Number(p.usdEquiv || 0), 0); totalPagado += pagadoItem; // Si queda deuda en este mes específico if ((montoAsignadoItem - pagadoItem) > 0.01) { mesesAdeudadosCount++; const [yyyy, mm] = (exp.period || toInput).split('-'); const nombreMes = new Date(yyyy, mm - 1, 1).toLocaleString('es', { month: 'long', year: 'numeric' }).toUpperCase(); conceptosPendientes.push(nombreMes); } }); const diferencia = totalAsignado - totalPagado; let estadoTexto = ''; let estadoClase = ''; if (diferencia > 0.01) { estadoTexto = `Deuda: $${diferencia.toFixed(2)}`; estadoClase = 'text-rose-500 font-bold'; } else if (diferencia < -0.01) { estadoTexto = `A favor: $${Math.abs(diferencia).toFixed(2)}`; estadoClase = 'text-blue-500 font-bold'; } else { estadoTexto = 'Solvente ($0.00)'; estadoClase = 'text-emerald-500 font-bold'; } const detalleTexto = conceptosPendientes.length > 0 ? conceptosPendientes.join(', ') : 'Solvente'; ordBody.innerHTML += ` ${o.apto} ${mesesAdeudadosCount > 0 ? mesesAdeudadosCount : '-'} ${detalleTexto} $${totalAsignado.toFixed(2)} ${estadoTexto} `; }); } // ========================================== // SECCIÓN 2: REPORTE DE CUOTAS EXTRAORDINARIAS // ========================================== const extraBody = document.getElementById('report-extraordinary-body'); if (extraBody) { extraBody.innerHTML = ''; // Filtrar gastos extraordinarios en el rango seleccionado const extraordinaryExpenses = state.expenses.filter(e => { const isExtra = e.type === 'extraordinaria' || e.type === 'extraordinary'; if (!isExtra) return false; if (!e.period) return true; if (fromInput && e.period < fromInput) return false; if (toInput && e.period > toInput) return false; return true; }); if (extraordinaryExpenses.length === 0) { extraBody.innerHTML = `No hay cuotas extraordinarias registradas para este periodo.`; } state.owners.forEach(o => { const factor = o.alicuota > 1 ? o.alicuota / 100 : Number(o.alicuota || 0); let totalAsignadoExtra = 0; let totalPagadoExtra = 0; let conceptosPendientesExtra = []; let conceptosAdeudadosCount = 0; extraordinaryExpenses.forEach(exp => { const montoAsignadoItem = Number(exp.amount || 0) * factor; totalAsignadoExtra += montoAsignadoItem; const approvedExtraPayments = state.payments.filter(p => p.apto === o.apto && p.status === 'Aprobado' && (p.type === 'extraordinaria' || p.type === 'extraordinary') && (!p.expense_id || p.expense_id === exp.id) ); const pagadoItem = approvedExtraPayments.reduce((acc, p) => acc + Number(p.usdEquiv || 0), 0); totalPagadoExtra += pagadoItem; if ((montoAsignadoItem - pagadoItem) > 0.01) { conceptosAdeudadosCount++; conceptosPendientesExtra.push(exp.description || 'Cuota Extraordinaria'); } }); const diferenciaExtra = totalAsignadoExtra - totalPagadoExtra; let estadoExtraTexto = ''; let estadoExtraClase = ''; if (diferenciaExtra > 0.01) { estadoExtraTexto = `Deuda: $${diferenciaExtra.toFixed(2)}`; estadoExtraClase = 'text-rose-500 font-bold'; } else if (diferenciaExtra < -0.01) { estadoExtraTexto = `A favor: $${Math.abs(diferenciaExtra).toFixed(2)}`; estadoExtraClase = 'text-blue-500 font-bold'; } else { estadoExtraTexto = 'Solvente ($0.00)'; estadoExtraClase = 'text-emerald-500 font-bold'; } const detalleExtraTexto = conceptosPendientesExtra.length > 0 ? conceptosPendientesExtra.join(', ') : 'Solvente'; extraBody.innerHTML += ` ${o.apto} ${conceptosAdeudadosCount > 0 ? conceptosAdeudadosCount : '-'} ${detalleExtraTexto} $${totalAsignadoExtra.toFixed(2)} ${estadoExtraTexto} `; }); } } // 3. Reporte Proveedores const provBody = document.getElementById('report-providers-body'); if (provBody) { provBody.innerHTML = ''; state.providers.forEach(p => { provBody.innerHTML += ` ${p.name} ${p.service} ${p.phone || 'N/A'} `; }); } async function saveBuildingConfig(e) { e.preventDefault(); const name = document.getElementById('cfg-name').value; const rif = document.getElementById('cfg-rif').value; const address = document.getElementById('cfg-address').value; const { data, error } = await db.from('building_config').upsert([{ id: 1, name, rif, address }]); if (error) { alert('Error al guardar la configuración: ' + error.message); return; } await loadStorage(); alert('¡Configuración actualizada con éxito!'); } // ==================== CONFIGURACIÓN DE TIEMPO REAL (OPCIONAL) ==================== function setupRealtimeSubscriptions() { // Función de respaldo para evitar errores si no se usa Realtime console.log("Suscripciones en tiempo real omitidas."); } // ==================== INICIALIZACIÓN DE LA APP ==================== document.addEventListener('DOMContentLoaded', () => { loadStorage(); setupRealtimeSubscriptions(); });
